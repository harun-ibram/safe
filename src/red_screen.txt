// --- SPI & display functions (same as before) ---
void spi_init() {
    DDRB |= (1 << PB2) | (1 << PB3) | (1 << PB5);
    PORTB |= (1 << PB2);
    DDRB &= ~(1 << PB4);
    SPCR = (1 << SPE) | (1 << MSTR) | (1 << SPR0);
    SPSR &= ~(1 << SPI2X);
}

void spi_send(uint8_t byte) {
    SPDR = byte;
    while (!(SPSR & (1 << SPIF)));
}

void ili9341_cmd(uint8_t cmd) { DC_LOW();  spi_send(cmd);  }
void ili9341_data(uint8_t d)  { DC_HIGH(); spi_send(d);    }

void ili9341_reset() {
    RES_HIGH(); _delay_ms(10);
    RES_LOW();  _delay_ms(20);
    RES_HIGH(); _delay_ms(150);
}

void ili9341_init() {
    ili9341_reset();
    ili9341_cmd(0x01); _delay_ms(150);
    ili9341_cmd(0x11); _delay_ms(120);
    ili9341_cmd(0x36); ili9341_data(0x48);
    ili9341_cmd(0x3A); ili9341_data(0x55);
    ili9341_cmd(0x29); _delay_ms(50);
}

// --- Set a window region to write pixels into ---
void ili9341_set_window(uint16_t x0, uint16_t y0, uint16_t x1, uint16_t y1) {
    ili9341_cmd(0x2A);
    ili9341_data(x0 >> 8); ili9341_data(x0 & 0xFF);
    ili9341_data(x1 >> 8); ili9341_data(x1 & 0xFF);
    ili9341_cmd(0x2B);
    ili9341_data(y0 >> 8); ili9341_data(y0 & 0xFF);
    ili9341_data(y1 >> 8); ili9341_data(y1 & 0xFF);
    ili9341_cmd(0x2C);
}

// --- Fill screen with solid color ---
void ili9341_fill(uint16_t color) {
    ili9341_set_window(0, 0, 239, 319);
    DC_HIGH();
    uint8_t hi = color >> 8, lo = color & 0xFF;
    for (uint32_t i = 0; i < 76800; i++) {
        spi_send(hi);
        spi_send(lo);
    }
}

// --- Draw a single character at (x, y) with scale factor ---
// fg/bg are RGB565 colors
void ili9341_draw_char(uint16_t x, uint16_t y, char c,
                       uint16_t fg, uint16_t bg, uint8_t scale) {
    if (c < 32 || c > 126) c = '?';
    uint8_t col_data[5];
    for (uint8_t i = 0; i < 5; i++)
        col_data[i] = pgm_read_byte(&font5x8[c - 32][i]);

    // Each char is 5 wide + 1 gap, 8 tall
    ili9341_set_window(x, y,
                       x + (6 * scale) - 1,
                       y + (8 * scale) - 1);
    DC_HIGH();
    uint8_t fg_hi = fg >> 8, fg_lo = fg & 0xFF;
    uint8_t bg_hi = bg >> 8, bg_lo = bg & 0xFF;

    for (uint8_t row = 0; row < 8; row++) {
        for (uint8_t sr = 0; sr < scale; sr++) {
            for (uint8_t col = 0; col < 5; col++) {
                uint8_t pixel_on = (col_data[col] >> row) & 1;
                for (uint8_t sc = 0; sc < scale; sc++) {
                    if (pixel_on) {
                        spi_send(fg_hi); spi_send(fg_lo);
                    } else {
                        spi_send(bg_hi); spi_send(bg_lo);
                    }
                }
            }
            // 1-pixel column gap (background colored)
            for (uint8_t sc = 0; sc < scale; sc++) {
                spi_send(bg_hi); spi_send(bg_lo);
            }
        }
    }
}

// --- Draw a null-terminated string ---
void ili9341_draw_string(uint16_t x, uint16_t y, const char *str,
                         uint16_t fg, uint16_t bg, uint8_t scale) {
    while (*str) {
        ili9341_draw_char(x, y, *str++, fg, bg, scale);
        x += 6 * scale;  // Advance by char width + gap
        if (x + 6 * scale > 240) break; // Don't draw off screen
    }
}

// --- Draw an integer ---
void ili9341_draw_int(uint16_t x, uint16_t y, int32_t val,
                      uint16_t fg, uint16_t bg, uint8_t scale) {
    char buf[12];
    // Manual int-to-string (no sprintf to save flash)
    uint8_t i = 0;
    uint8_t negative = 0;
    if (val < 0) { negative = 1; val = -val; }
    if (val == 0) { buf[i++] = '0'; }
    while (val > 0) { buf[i++] = '0' + (val % 10); val /= 10; }
    if (negative) buf[i++] = '-';
    // Reverse
    for (uint8_t a = 0, b = i - 1; a < b; a++, b--) {
        char tmp = buf[a]; buf[a] = buf[b]; buf[b] = tmp;
    }
    buf[i] = '\0';
    ili9341_draw_string(x, y, buf, fg, bg, scale);
}

// Common RGB565 colors
#define BLACK   0x0000
#define WHITE   0xFFFF
#define RED     0xF800
#define GREEN   0x07E0
#define BLUE    0x001F
#define YELLOW  0xFFE0
#define CYAN    0x07FF
#define MAGENTA 0xF81F

int main(void) {
    RES_DDR |= (1 << RES_PIN);
    DC_DDR  |= (1 << DC_PIN);

    spi_init();
    ili9341_init();

    // Red background
    ili9341_fill(RED);

    // Draw text at scale 1 (5x8 px), 2 (10x16 px), 3 (15x24 px)...
    ili9341_draw_string(10, 10,  "Hello, World!", WHITE, RED, 1);
    ili9341_draw_string(10, 30,  "Hello, World!", WHITE, RED, 2);
    ili9341_draw_string(10, 70,  "Hello, World!", WHITE, RED, 3);

    // Draw numbers
    ili9341_draw_int(10, 130, 12345,   YELLOW, RED, 2);
    ili9341_draw_int(10, 160, -99,     CYAN,   RED, 2);
    ili9341_draw_int(10, 190, 0,       WHITE,  RED, 2);

    while (1) {}
}
